Assignment 4  (DQL)

Q1) Display the names (first_name, last_name) using aliases "First Name", "Last Name".

	select first_name as 'First Name'
               last_name as 'Last Name'
 	from employees;

Q2) Get unique department ID from employee table.
	
	select distinct department_id 
	from employees;

Q3) Get all employee details ordered by first name in descending order.

	select * from employee
	order by first_name desc;

Q4) Get first name, last name, salary and PF of all employees (PF is 15% of salary).

	select first_name, last_name, salary * 0.15 as PF
	from employee;

Q5) Get employee ID, names and salary in ascending order of salary.

	select emp_id, first_name, salary
	from employee
	order by salary;

Q6) Get the total salaries payable to employees.

	select sum(salary) as payable
	from employees;

Q7) Get the maximum and minimum salary from employees table.

	select 
	       max(salary) as maximum_salary,
	       min(salary) as minimum_salary
	from employees;

Q8) Get the average salary and number of employees in employees table.

	select avg(salary) as average_salary,
		count(*) as number_emp_cnt
	from employees;


Q9) Get the number of employees working with the company.
	
	select count(*) as total_emp
	from employees;

Q10) Get the number of jobs available in the employees table.

	select count(distinct job_id) as total_jobs
	from employee;

Q11) Select first 10 records from a table.

	select * from employee
	limit 10;

Q12) Display first name, last name and salary for employees whose salary is not in the 	      range 10000 through 15000.


	select first_name, last_name, salary
	from employees
	where salary not between 10000 and 15000;

Q13) Display first name, last name and department ID of employees in departments 30 or 100 in ascending order.

	select first_name, last_name, department_id
	from employees
	where department_id in(30, 100)
	order by department_id asc;


Q14) Display first name, last name and salary for employees whose salary is not in the range 10000 through 15000 and are in department 30 or 100.

	select first_name, last_name, department_id
	from employees
	where salary not between 1000 and 15000  and department_id in (30, 100);

Q15) Display first name, last name and hire date for employees hired in 1987.

	SELECT
    		first_name,
    		last_name,
    		hire_date
	FROM employees
	WHERE YEAR(hire_date) = 1987;

Q16) Display first name of employees who have both "b" and "c" in their first name.

	SELECT first_name
	FROM employees
	WHERE first_name LIKE '%b%' AND first_name LIKE '%c%';


Q18) Display last name of employees whose names have exactly 6 characters.

	select last_name
	from employees
	where length(last_name) = 6;

Q19) Display last name of employees having 'e' as the third character.

	select last_name
	from employees
	where last_name like '___e%';
