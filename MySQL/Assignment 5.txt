Assignment 5 (More DQL)

Q1) Books written by Loni with price less than 600
	
	select * from books
	where Author_name = 'Loni' and cost < 600;

2) Issue details for books that are not returned yet.

	select * from books
	where return_date is null;

3) Update blank return dates to 31-Dec-06, excluding 7005 and 7006

	update lib_issue
	set return_date = '2006-12-31'
	where return_date is null and lib_isssue_id not in(7005, 7006)

4) Issue details for books issued for more than 30 days.

	select * from lib_issue
	where datediff(return_date, issue_date) > 30;

5) Books with cost between 500 and 750 and category Database

	select * from books
	where cost between 500 and 750 and category = 'Database';

6) Books belonging to Science, Database, Fiction or Management

	select * from books
	where category in('Science', 'Database', 'Fiction', 'Management');


7) Members in descending order of penalty.

	select * from member
	order by penalty_amount desc;

8) Modify book 103 cost to 300 and category to RDBMS

	update books
	set cost = 300, category = 'RDBMS'
	where book_no = 103;

9) Books in ascending category and descending cost.

	select * from books
	order by category asc, cost desc;

10) Books containing the word SQL in their name

	select * from books
	where book_name like '%SQL%';

11) Display issue ID, issue date, return date and number of days issued.

	select issue_id, issue_date, return_date, 
	datediff(return_date, issue_date) as no_of_days
	from lib_issue;

13) Count total books issued to Member 101.

	select count(*) as total_issue_books
	from lib_issue
	where member_id = 101;

14) Total penalty due for all members

	select sum(penalty_amount) as total_penalty
	from member;

15) Total number of members

	select count(*) as total_member
	from member;


20) Maximum, minimum, total and average penalty.

	select max(penalty_amount) as maximum,
		min(penalty_amount) as minimum,
		sum(penalty_amount) as total,
		avg(penalty_amount) as average
	from member;
